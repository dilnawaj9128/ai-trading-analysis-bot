from unittest.mock import Mock, patch

import pandas as pd

def test_credit_snapshot_returned():
    config = create_research_config(
        "quotex", "forex", "EUR/USD", "1min", 15
    )

    response = Mock()
    response.raise_for_status.return_value = None
    response.headers = {
        "api-credits-used": "1",
        "api-credits-left": "799",
    }

    response.json.return_value = {
        "meta": {
            "symbol": "EUR/USD",
            "interval": "1min",
        },
        "values": [
            {
                "datetime": "2026-11-12 10:00:00",
                "open": "1.1000",
                "high": "1.1010",
                "low": "1.0990",
                "close": "1.1005",
            }
        ],
    }

    session = Mock()
    session.get.return_value = response

    with patch("market_data.historical_collector.reserve_request"):
        result = fetch_historical_candles(
            config,
            start=pd.Timestamp("2026-11-12T10:00:00Z"),
            end=pd.Timestamp("2026-11-12T10:01:00Z"),
            now=pd.Timestamp("2026-11-13T12:00:00Z"),
            api_key="test-key",
            session=session,
            outputsize=10,
        )

    snapshot = result["credit_snapshot"]

    assert snapshot.credits_used == 1
    assert snapshot.credits_left == 799
    assert snapshot.information_available is True

========== tests/test_historical_http_integration.py ==========
from unittest.mock import Mock, patch

import pandas as pd
import pytest
import requests

from config.market_config import create_research_config
from market_data.historical_collector import fetch_historical_candles
from market_data.twelve_data_client import MarketDataError


START = pd.Timestamp("2026-11-12T10:00:00Z")
END = pd.Timestamp("2026-11-12T10:02:00Z")
NOW = pd.Timestamp("2026-11-13T12:00:00Z")


@pytest.fixture
def config():
    return create_research_config(
        "quotex",
        "forex",
        "EUR/USD",
        "1min",
        15,
    )


def run_fetch(config, response):
    session = Mock()
    session.get.return_value = response

    with patch(
        "market_data.historical_collector.reserve_request"
    ) as reserve:
        try:
            result = fetch_historical_candles(
                config,
                start=START,
                end=END,
                now=NOW,
                api_key="test-key",
                session=session,
                outputsize=10,
            )
            return result, session, reserve
        except Exception:
            raise


@pytest.mark.parametrize("status_code", [429, 500])
def test_http_errors_are_rejected(config, status_code):
    response = Mock()

    response.raise_for_status.side_effect = requests.HTTPError(
        f"HTTP {status_code}",
        response=response,
    )

    response.status_code = status_code

    with pytest.raises(
        MarketDataError,
        match="Historical request failed",
    ):
        run_fetch(config, response)


def test_valid_http_response(config):
    response = Mock()
    response.headers = {}
    response.raise_for_status.return_value = None

    response.json.return_value = {
        "meta": {
            "symbol": "EUR/USD",
            "interval": "1min",
        },
        "values": [
            {
                "datetime": "2026-11-12 10:00:00",
                "open": "1.1000",
                "high": "1.1010",
                "low": "1.0990",
                "close": "1.1005",
            },
            {
                "datetime": "2026-11-12 10:01:00",
                "open": "1.1005",
                "high": "1.1020",
                "low": "1.1000",
                "close": "1.1015",
            },
        ],
    }

    result, session, reserve = run_fetch(config, response)

    assert result["candles"] == 2
    assert result["possibly_truncated"] is False
    assert result["analysis_only"] is True
    assert result["execution_enabled"] is False

    reserve.assert_called_once()
    session.get.assert_called_once()


def test_provider_error_payload(config):
    response = Mock()
    response.headers = {}
    response.raise_for_status.return_value = None

    response.json.return_value = {
        "status": "error",
        "code": 429,
        "message": "API credits exhausted",
    }

    with pytest.raises(
        MarketDataError,
        match="Provider error",
    ):
        run_fetch(config, response)
========== tests/test_invalid_candle_reservation.py ==========
from datetime import datetime, timezone
from types import SimpleNamespace
from unittest.mock import patch

import pandas as pd
import pytest

from config.market_config import create_research_config
from market_data.historical_collector import fetch_historical_candles
from market_data.twelve_data_client import MarketDataError


def test_invalid_candle_marks_reservation_uncertain():
    config = create_research_config(
        "quotex", "forex", "EUR/USD", "1min", 15
    )

    now = pd.Timestamp(
        datetime(2026, 11, 13, 12, 0, tzinfo=timezone.utc)
    )

    invalid_payload = {
        "meta": {
            "symbol": "EUR/USD",
            "interval": "1min",
        },
        "values": [
            {
                "datetime": "2026-11-13 12:05:00",
                "open": "1.1000",
                "high": "1.1010",
                "low": "1.0990",
                "close": "1.1005",
            }
        ],
    }

    with (
        patch(
            "market_data.historical_collector._validate_historical_request"
        ) as validate,
        patch(
            "market_data.historical_collector.authorize_request"
        ) as authorize,
        patch(
            "market_data.historical_collector._perform_historical_http_request"
        ) as transport,
        patch(
            "market_data.authorization_ledger.transition_authorization"
        ) as transition,
    ):
        validate.return_value = (
            now - pd.Timedelta(minutes=10),
            now,
            now,
            pd.Timedelta(minutes=1),
            "fake-test-key",
        )

        authorize.return_value = SimpleNamespace(
            request_id="invalid-candle-request"
        )

        transport.return_value = (
            invalid_payload,
            None,
            SimpleNamespace(status_code=200, headers={}),
        )

        with pytest.raises(
            MarketDataError,
            match="Candle outside requested window",
        ):
            fetch_historical_candles(
                config,
                start=now - pd.Timedelta(minutes=10),
                end=now,
                api_key="fake-test-key",
                provider_credit_path="fake-provider.json",
            )

        transition.assert_called_once_with(
            provider_state_path="fake-provider.json",
            request_id="invalid-candle-request",
            status="UNCERTAIN",
        )

from unittest.mock import Mock, patch

import pandas as pd
import pytest
import requests

from config.market_config import create_research_config
from market_data.historical_collector import fetch_historical_candles
from market_data.twelve_data_client import MarketDataError


@pytest.fixture
def config():
    return create_research_config(
        "quotex", "forex", "EUR/USD", "1min", 15
    )


def execute(config, response):
    session = Mock()
    session.get.return_value = response

    with patch("market_data.historical_collector.reserve_request"):
        return fetch_historical_candles(
            config,
            start=pd.Timestamp("2026-11-12T10:00:00Z"),
            end=pd.Timestamp("2026-11-12T10:02:00Z"),
            now=pd.Timestamp("2026-11-13T12:00:00Z"),
            api_key="test-key",
            session=session,
            outputsize=10,
        )


@pytest.mark.parametrize(
    "status,expected",
    [
        (429, "HTTP 429 rate limit exceeded"),
        (500, "HTTP 500 provider server error"),
        (503, "HTTP 503 provider server error"),
        (403, "HTTP 403"),
    ],
)
def test_http_status_classification(config, status, expected):
    response = Mock()
    response.status_code = status
    response.raise_for_status.side_effect = requests.HTTPError(
        f"HTTP {status}",
        response=response,
    )

    with pytest.raises(MarketDataError, match=expected):
        execute(config, response)


def test_invalid_json_classification(config):
    response = Mock()
    response.raise_for_status.return_value = None
    response.json.side_effect = ValueError("Invalid JSON")

    with pytest.raises(MarketDataError, match="invalid JSON"):
        execute(config, response)


def test_network_error_classification(config):
    session = Mock()
