def fetch_bootstrap_historical_candles(
    config,
    *,
    start,
    end,
    approval_id,
    explicitly_approved=False,
    api_key=None,
    state_path=STATE_PATH,
    provider_credit_path=None,
    approval_path="data/api_usage/bootstrap_claim.json",
    now=None,
    session=None,
    outputsize=MAX_HISTORY_BATCH,
):
    """One-time approved historical bootstrap; research-only."""

    from market_data.bootstrap_request_coordinator import execute_bootstrap
    from market_data.provider_credit_state import DEFAULT_PATH

    if provider_credit_path is None:
        provider_credit_path = DEFAULT_PATH

    # Validate before consuming approval or local request budget.
    start, end, now, step, key = _validate_historical_request(
        config,
        start=start,
        end=end,
        api_key=api_key,
        now=now,
        outputsize=outputsize,
    )

    captured = {}

    def request_callback():
        payload, snapshot, response = _perform_historical_http_request(
            config,
            start=start,
            end=end,
            step=step,
            key=key,
            outputsize=outputsize,
            session=session,
            provider_credit_path=None,
            now=now,
        )
        captured["payload"] = payload
        captured["snapshot"] = snapshot
        return response

    def validate_response(_response):
        captured["frame"] = parse_historical_payload(
            captured["payload"],
            symbol=config.symbol,
            interval=config.timeframe,
            start=start,
            end=end,
            now=now,
        )

    bootstrap = execute_bootstrap(
        request_callback=request_callback,
        response_validator=validate_response,
        approval_id=approval_id,
        explicitly_approved=explicitly_approved,
        approval_path=approval_path,
        provider_state_path=provider_credit_path,
        local_budget_path=state_path,
        now=now.to_pydatetime(),
    )

    frame = captured["frame"]

    return {
        "frame": frame,
        "credit_snapshot": captured["snapshot"],
        "bootstrap": bootstrap,
        "symbol": config.symbol,
        "interval": config.timeframe,
        "start_utc": start.isoformat(),
        "end_utc": end.isoformat(),
        "candles": len(frame),
        "gaps": int((frame["timestamp"].diff().dropna() > step).sum()),
        "possibly_truncated": len(frame) >= outputsize,
        "analysis_only": True,
        "execution_enabled": False,
    }
=== Bootstrap approval implementation ===
"""Persistent, one-time bootstrap approval claim.

No HTTP requests, trading execution, or API budget reservations.
"""

import json
import os
from datetime import datetime, timezone
from pathlib import Path


class BootstrapApprovalRequired(RuntimeError):
    pass


class BootstrapApprovalAlreadyUsed(RuntimeError):
    pass


def claim_bootstrap_approval(
    *,
    approval_id,
    explicitly_approved=False,
    path="data/api_usage/bootstrap_claim.json",
    now=None,
):
    if not isinstance(approval_id, str) or not approval_id.strip():
        raise ValueError("Non-empty approval_id required")

    if type(explicitly_approved) is not bool:
        raise ValueError("explicitly_approved must be boolean")

    if not explicitly_approved:
        raise BootstrapApprovalRequired(
            "Explicit bootstrap approval required"
        )

    now = now if now is not None else datetime.now(timezone.utc)

    if not isinstance(now, datetime) or now.tzinfo is None:
        raise ValueError("Timezone-aware datetime required")

    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    payload = {
        "version": 1,
        "approval_id": approval_id,
        "claimed_at": now.astimezone(timezone.utc).isoformat(),
        "status": "claimed",
    }

    # O_EXCL makes file creation atomic across processes.
    flags = os.O_WRONLY | os.O_CREAT | os.O_EXCL

    try:
        fd = os.open(path, flags, 0o600)
    except FileExistsError as exc:
        raise BootstrapApprovalAlreadyUsed(
            "Bootstrap approval already claimed"
        ) from exc

    # Fail closed: if writing fails, retain the claim file.
    with os.fdopen(fd, "w", encoding="utf-8") as handle:
        json.dump(payload, handle)
        handle.flush()
        os.fsync(handle.fileno())

    return payload
=== Bootstrap policy ===
"""Provider credit bootstrap policy.

Decision-only module. No HTTP calls or local budget reservations.
"""

from dataclasses import dataclass
from datetime import datetime, timezone

from market_data.provider_credit_guard import (
    ProviderCreditLimitReached,
    ProviderCreditStatusUnknown,
)
from market_data.provider_credit_state import (
    DEFAULT_PATH,
    load_credit_state,
)
from market_data.pre_request_credit_gate import check_before_request


@dataclass(frozen=True)
class BootstrapDecision:
    action: str
    reason: str
    requires_explicit_approval: bool


def decide_credit_bootstrap(
    *,
    path=DEFAULT_PATH,
    now=None,
    max_age_seconds=300,
    bootstrap_approved=False,
):
    if type(bootstrap_approved) is not bool:
        raise ValueError("Invalid bootstrap_approved")

    if now is None:
        now = datetime.now(timezone.utc)

    snapshot = load_credit_state(
        path=path,
        now=now,
        max_age_seconds=max_age_seconds,
    )

    if snapshot is None or snapshot.credits_left is None:
        if bootstrap_approved:
            return BootstrapDecision(
                action="BOOTSTRAP",
                reason="One explicitly approved bootstrap request may be attempted",
                requires_explicit_approval=False,
            )

        return BootstrapDecision(
            action="REQUIRE_APPROVAL",
            reason="Provider credits unknown or expired",
            requires_explicit_approval=True,
        )

    try:
        check_before_request(
            path=path,
            now=now,
            max_age_seconds=max_age_seconds,
        )
    except ProviderCreditLimitReached:
        return BootstrapDecision(
            action="BLOCK",
            reason="Insufficient reported provider credits",
            requires_explicit_approval=False,
        )
    except ProviderCreditStatusUnknown:
        return BootstrapDecision(
            action="REQUIRE_APPROVAL",
            reason="Provider credit status unavailable",
            requires_explicit_approval=True,
        )

    return BootstrapDecision(
        action="NORMAL",
        reason="Fresh provider credit snapshot permits normal requests",
        requires_explicit_approval=False,
    )

import json
from datetime import datetime, timezone
from unittest.mock import Mock

import pytest

from config.market_config import create_research_config
from market_data.historical_collector import (
    fetch_bootstrap_historical_candles,
)
from market_data.bootstrap_request_coordinator import BootstrapNotPermitted


NOW = datetime(2026, 11, 13, 12, 0, tzinfo=timezone.utc)


def config():
    return create_research_config(
        "quotex", "forex", "EUR/USD", "1min", 15
    )


def response():
    result = Mock()
    result.status_code = 200
    result.raise_for_status.return_value = None
    result.headers = {"api-credits-left": "99"}
    result.json.return_value = {
        "meta": {"symbol": "EUR/USD", "interval": "1min"},
        "values": [{
            "datetime": "2026-11-12 10:00:00",
            "open": "1.1000",
            "high": "1.1010",
            "low": "1.0990",
            "close": "1.1005",
        }],
    }
    return result


def execute(tmp_path, session, **overrides):
    args = {
        "start": "2026-11-12T10:00:00Z",
        "end": "2026-11-12T10:01:00Z",
        "approval_id": "test-approval",
        "explicitly_approved": True,
        "api_key": "mock-key",
        "state_path": tmp_path / "budget.json",
        "provider_credit_path": tmp_path / "provider.json",
        "approval_path": tmp_path / "approval.json",
        "now": NOW,
        "session": session,
        "outputsize": 10,
    }
    args.update(overrides)
    return fetch_bootstrap_historical_candles(config(), **args)


def test_bootstrap_fetches_and_validates_candles(tmp_path):
    session = Mock()
    session.get.return_value = response()

    result = execute(tmp_path, session)

    assert result["candles"] == 1
    assert result["analysis_only"] is True
    assert result["execution_enabled"] is False
    assert result["bootstrap"].credits_refreshed is True

    session.get.assert_called_once()

    budget = json.loads((tmp_path / "budget.json").read_text())
    assert budget["minute_used"] == 1
    assert budget["day_used"] == 1
