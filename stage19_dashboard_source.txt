"""Local read-only historical research dashboard."""

from pathlib import Path

import pandas as pd
import streamlit as st

from dashboard.report_loader import (
    DashboardReportError,
    dashboard_summary,
    hourly_chart_data,
    load_report,
    volatility_chart_data,
)


DEFAULT_REPORT = (
    "reports/eurusd_advanced_20261112.json"
)


def main():
    st.set_page_config(
        page_title="AI Trading Research Dashboard",
        page_icon="📊",
        layout="wide",
    )

    st.title("AI Trading Research Dashboard")
    st.caption(
        "Historical EUR/USD research | UTC | Read-only"
    )

    st.warning(
        "ANALYSIS ONLY — No trading signals, "
        "broker execution, or directional backtesting. "
        "Feed provenance remains unverified."
    )

    with st.sidebar:
        st.header("Research report")

        report_path = st.text_input(
            "Local JSON report path",
            value=DEFAULT_REPORT,
        )

        st.info(
            "This dashboard only reads an existing "
            "schema v3 research report."
        )

    try:
        report = load_report(Path(report_path))
        summary = dashboard_summary(report)
    except DashboardReportError as exc:
        st.error(f"Report validation failed: {exc}")
        st.stop()

    st.subheader("Dataset overview")

    columns = st.columns(4)

    columns[0].metric(
        "Accepted sessions",
        summary["accepted_sessions"],
    )
    columns[1].metric(
        "Unique candles",
        summary["unique_candles"],
    )
    columns[2].metric(
        "Duplicates",
        summary["duplicates"],
    )
    columns[3].metric(
        "Overlaps",
        summary["overlaps"],
    )

    coverage = report[
        "advanced_analytics"
    ]["coverage"]

    st.subheader("Historical data coverage")

    left, right = st.columns(2)

    with left:
        percentage = summary["coverage_percent"]

        st.metric(
            "Observed-window coverage",
            (
                f"{percentage:.2f}%"
                if percentage is not None
                else "N/A"
            ),
        )

        st.metric(
            "Missing minutes",
            summary["missing_minutes"],
        )

    with right:
        st.write(
            "First candle:",
            coverage["first_timestamp"],
        )
        st.write(
            "Last candle:",
            coverage["last_timestamp"],
        )

        st.caption(
            "Missing minutes are calculated between "
            "the first and last accepted timestamps. "
            "Market closures are not excluded."
        )

    st.subheader("Hourly candle behavior")

    hourly = pd.DataFrame(
        hourly_chart_data(report)
    )

    if hourly.empty:
        st.info("No accepted hourly data.")
    else:
        st.bar_chart(
            hourly.set_index("hour")[
                ["average_range"]
            ]
        )

        st.dataframe(
            hourly,
            width="stretch",
            hide_index=True,
        )

    st.subheader("Historical volatility ranking")

    ranking = pd.DataFrame(
        volatility_chart_data(report)
    )

    if ranking.empty:
        st.info("No accepted sessions to rank.")
    else:
        st.bar_chart(
            ranking.set_index("session")[
                ["average_range"]
            ]
        )

        st.dataframe(
            ranking,
            width="stretch",
            hide_index=True,
        )

    st.subheader("Session quality and selection")

    sessions = report.get("sessions", [])

    if sessions:
        session_rows = []

        for item in sessions:
            session_rows.append({
                "Dataset": item.get("dataset"),
                "Quality": item.get("quality", {}).get(
                    "status", "UNKNOWN"
                ) if isinstance(
                    item.get("quality"), dict
                ) else "See report",
                "Selection": item.get(
                    "selection_status"
                ),
                "Reason": item.get(
                    "selection_reason"
                ),
            })

        st.dataframe(
            pd.DataFrame(session_rows),
            width="stretch",
            hide_index=True,
        )
    else:
        st.info("No session records.")

    with st.expander("Research limitations"):
        for limitation in report.get(
            "limitations", []
        ):
            st.write("-", limitation)

    st.divider()

    st.caption(
        "Research-only visualization. "
        "Historical volatility does not predict "
        "future price direction."
    )


if __name__ == "__main__":
    main()

def fmt(value, digits=5):
    try:
        return "N/A" if value is None or pd.isna(value) else f"{float(value):.{digits}f}"
    except (ValueError, TypeError):
        return "N/A"


@st.fragment(run_every="10s")
def live_panel():
    if not st.session_state.get("simple_running"):
        return
    try:
        outcome = run_monitor_poll(
            config,
            interval_minutes=refresh_minutes,
            fetcher=fetch_authorized_live_candles,
        )
    except (MarketDataError, ValueError, OSError) as exc:
        st.error(f"Monitoring unavailable: {exc}")
        return

    status = outcome["status"]
    if status == "SUCCESS":
        st.session_state["simple_result"] = outcome["result"]
        st.session_state["simple_last_fetch"] = datetime.now(timezone.utc).isoformat()
    elif status == "WAITING":
        st.caption("Next permitted fetch: " + outcome["next_allowed_utc"])
    elif status == "BUDGET_BLOCKED":
        st.error("Local API request budget reached. Fetching paused.")
    else:
        st.error("Fetch failed: " + outcome.get("reason", "Unknown error"))

    result = st.session_state.get("simple_result")
    if result is None:
        st.info("Waiting for the first successful API fetch.")
        return
    try:
        analysis = analyze_live_candles(result["frame"])
        enriched = analysis["frame"]
        latest = analysis["latest"]
        timestamp = pd.Timestamp(result["latest_utc"])
        if timestamp.tzinfo is None:
            raise ValueError("Latest candle has no timezone")
        age = max(0, (datetime.now(timezone.utc) - timestamp.to_pydatetime()).total_seconds())
        stale = age > {"1min": 60, "5min": 300, "15min": 900}[timeframe] * 3
        gaps = result["gaps"]
    except (ValueError, TypeError, KeyError) as exc:
        st.error(f"Analysis unavailable: {exc}")
        return

    if stale or gaps or status != "SUCCESS":
        st.warning("Caution: " + "; ".join([
            *(["latest candle is stale"] if stale else []),
            *([f"{gaps} missing candle interval(s)"] if gaps else []),
            *(["showing last successful fetch"] if status != "SUCCESS" else []),
        ]))

    smart = assess_market(analysis, stale=stale, gaps=gaps)

    # Save history only for a newly successful fetch, never on WAITING
    # fragment reruns. The database operation makes no API requests.
    if status == "SUCCESS":
        try:
            record_id = save_analysis(
                symbol=config.symbol,
                timeframe=config.timeframe,
                analysis=analysis,
                assessment=smart,
            )
            st.caption(f"Analysis history saved (record #{record_id}).")
        except (OSError, ValueError, TypeError, KeyError, sqlite3.Error) as exc:
            st.warning(f"Could not save analysis history: {exc}")
    st.subheader("Smart Market Assessment")
    st.metric("Market condition", smart["condition"])
    st.info(smart["explanation"])
    st.caption("Research-only indicator alignment, not an expiry or directional prediction.")

    trend = "Mixed / neutral"
    e9, e21 = latest.get("EMA_9"), latest.get("EMA_21")
    if pd.notna(e9) and pd.notna(e21):
        trend = "EMA 9 above EMA 21" if e9 > e21 else ("EMA 9 below EMA 21" if e9 < e21 else "EMAs equal")
    a, b, c = st.columns(3)
    a.metric("Latest close", fmt(latest.get("Close")))
    b.metric("EMA observation", trend)
    c.metric("Data age", f"{age / 60:.1f} min")

    fig = go.Figure()
    fig.add_trace(go.Candlestick(x=enriched["timestamp"], open=enriched["Open"], high=enriched["High"], low=enriched["Low"], close=enriched["Close"], name="Price"))
    for key, name in (("EMA_9", "EMA 9"), ("EMA_21", "EMA 21")):
        fig.add_trace(go.Scatter(x=enriched["timestamp"], y=enriched[key], mode="lines", name=name))
    fig.update_layout(title=f"{symbol} • {timeframe} completed candles", height=470, xaxis_rangeslider_visible=False, margin=dict(l=15, r=15, t=45, b=20))
    st.plotly_chart(fig, width="stretch")

    st.subheader("What the indicators say")
    rsi = latest.get("RSI_14")
    rsi_text = "Not enough data" if pd.isna(rsi) else ("Above 70: high RSI reading" if rsi > 70 else ("Below 30: low RSI reading" if rsi < 30 else "Between 30–70: middle range"))
    st.write(f"**RSI 14:** {fmt(rsi, 1)} — {rsi_text}")
    st.write(f"**EMA 9 / 21:** {trend}. This describes the recent price trend, not the next candle.")
    for observation in analysis["observations"]:
        st.caption("• " + observation)
    st.caption("These observations are not UP/DOWN predictions or trading recommendations.")

    with st.expander("Advanced details (MACD, ATR, patterns, raw candles, credits)"):
        st.write("**ATR 14:**", fmt(latest.get("ATR_14")))
        macd_cols = [col for col in enriched.columns if "MACD" in col.upper()]
        if macd_cols:
            st.write("**MACD — latest candles**")
            st.dataframe(enriched[["timestamp"] + macd_cols].tail(5), width="stretch")
        levels = [col for col in enriched.columns if "support" in col.lower() or "resistance" in col.lower()]
        if levels:
            st.write("**Support / resistance**")
            st.dataframe(enriched[["timestamp"] + levels].tail(3), width="stretch")
        patterns = analysis["patterns"]
        cols = ["Pattern", "Pattern_Direction", "Pattern_Description"]
        if all(col in patterns.columns for col in cols):
            st.write("**Recent candlestick patterns**")
            st.dataframe(patterns[["timestamp"] + cols].tail(5), width="stretch")
        st.write("**Recent completed candles**")
        st.dataframe(result["frame"].tail(5), width="stretch")
        try:
            st.write("**Scheduler:**", read_monitor_state().get("last_status", "Unknown"))
        except (OSError, ValueError, MarketDataError):
            st.write("Scheduler state unavailable")
        try:
            if STATE_PATH.exists():
                usage = json.loads(STATE_PATH.read_text(encoding="utf-8"))
                today = datetime.now(timezone.utc).strftime("%Y-%m-%d")

                )
            else:
                st.caption("No saved analysis records yet.")
        except (OSError, ValueError, sqlite3.Error) as exc:
            st.caption(f"History unavailable: {exc}")

========== dashboard/chart_data.py ==========

"""Verified, read-only historical candle chart preparation."""

from __future__ import annotations

from pathlib import Path

import pandas as pd

from dashboard.report_loader import (
    DashboardReportError,
    load_report,
)
from data.dataset_quality import audit_dataset
from simulator.session_research import (
    CSV_NAME,
    RECEIPT_NAME,
)


COLUMNS = ("timestamp", "Open", "High", "Low", "Close")


def accepted_datasets(report):
    """Only use datasets accepted by the session selector."""

    sessions = report.get("sessions", [])

    if not isinstance(sessions, list):
        raise DashboardReportError("Invalid session records")

    selected = []

    for session in sessions:
        if not isinstance(session, dict):
            raise DashboardReportError("Invalid session entry")

        if session.get("selection_status") == "ACCEPTED":
            directory = session.get("dataset")

            if not isinstance(directory, str) or not directory:
                raise DashboardReportError(
                    "Accepted session has no dataset path"
                )

            selected.append(directory)

    if len(selected) != report["datasets_accepted"]:
        raise DashboardReportError(
            "Accepted dataset count mismatch"
        )

    return selected


def load_chart_candles(dataset_dir):
    """
    Re-audit original CSV and receipt before charting.

    No file modifications, no broker access.
    """

    directory = Path(dataset_dir)

    csv_path = directory / CSV_NAME
    receipt_path = directory / RECEIPT_NAME

    audit = audit_dataset(csv_path, receipt_path)

    if audit.get("status") != "PASS":
        raise DashboardReportError(
            "Dataset failed quality re-audit"
        )

    frame = pd.read_csv(csv_path)

    if not set(COLUMNS).issubset(frame.columns):
        raise DashboardReportError(
            "Missing OHLC or timestamp columns"
        )

    frame = frame.loc[:, COLUMNS].copy()

    frame["timestamp"] = pd.to_datetime(
        frame["timestamp"],
        utc=True,
        errors="raise",
    )

    frame = frame.sort_values(
        "timestamp"
    ).reset_index(drop=True)

    return frame


def calculate_indicators(frame):
    """
    Historical EMA 9/21 and Wilder RSI 14.

    Warmup values remain unavailable (NaN).
    No directional classification is produced.
    """

    result = frame.copy()

    closes = result["Close"].astype(float)

    result["EMA_9"] = closes.ewm(
        span=9,
        adjust=False,
        min_periods=9,
    ).mean()

    result["EMA_21"] = closes.ewm(
        span=21,
        adjust=False,
        min_periods=21,
    ).mean()

    # Wilder RSI: initial 14-period mean, then
    # recursive smoothing of gains and losses.
    period = 14
    delta = closes.diff()
    gains = delta.clip(lower=0)
    losses = (-delta).clip(lower=0)

    rsi = pd.Series(
        float("nan"),
        index=result.index,
        dtype=float,
    )

    if len(result) > period:
        avg_gain = gains.iloc[1:period + 1].mean()
        avg_loss = losses.iloc[1:period + 1].mean()

        for index in range(period, len(result)):
            if index > period:
                avg_gain = (
                    avg_gain * (period - 1)
                    + float(gains.iloc[index])
                ) / period

                avg_loss = (
                    avg_loss * (period - 1)
                    + float(losses.iloc[index])
                ) / period

            if avg_loss == 0 and avg_gain == 0:
                value = 50.0
            elif avg_loss == 0:
                value = 100.0
            elif avg_gain == 0:
                value = 0.0
            else:
                relative_strength = avg_gain / avg_loss
                value = 100 - (
                    100 / (1 + relative_strength)
                )

            rsi.iloc[index] = value

    result["RSI_14"] = rsi

    return result


def filter_time_window(frame, start, end):
    """Inclusive UTC timestamp filter."""

    start_utc = pd.Timestamp(start)
    end_utc = pd.Timestamp(end)

    if start_utc.tzinfo is None:
        start_utc = start_utc.tz_localize("UTC")
    else:
        start_utc = start_utc.tz_convert("UTC")

    if end_utc.tzinfo is None:
        end_utc = end_utc.tz_localize("UTC")
    else:
        end_utc = end_utc.tz_convert("UTC")

    if start_utc > end_utc:
        raise ValueError("Start time must precede end time")

    return frame.loc[
        frame["timestamp"].between(
            start_utc, end_utc
        )
    ].copy()

from strategy.trend_analysis import analyze_trend
from strategy.market_filter import analyze_market_conditions


def generate_signal(
    data,
    expiry_minutes=15,
    data_mode="unknown",
    data_quality_status="INVALID",
):
    """
    Generate an explainable, analysis-only signal.

    Directional analysis is allowed only when:
    - Feed mode is explicitly live or delayed.
    - Data quality status is VALID.
    - At least 50 candles are available.

    Expiry is informational, not a prediction or guarantee.
    """

    if expiry_minutes not in (1, 5, 15):
        raise ValueError("expiry_minutes must be 1, 5, or 15.")

    mode = str(data_mode).strip().lower()
    quality_status = str(data_quality_status).strip().upper()

    def no_trade(reason, trend="UNKNOWN", momentum="UNKNOWN",
                 volatility="UNKNOWN", atr_percent=None):
        return {
            "signal": "NO TRADE",
            "trend": trend,
            "momentum": momentum,
            "volatility": volatility,
            "atr_percent": atr_percent,
            "expiry_minutes": expiry_minutes,
            "reasons": [reason],
        }

    # Safety gate 1: data quality
    if quality_status != "VALID":
        return no_trade(
            f"Data quality is {quality_status}; directional analysis blocked."
        )

    # Safety gate 2: verified feed mode
    if mode not in {"live", "delayed"}:
        return no_trade(
            "Data feed mode is unknown or unsupported. "
            "Verify the provider's feed mode before directional analysis."
        )

    # Safety gate 3: sufficient candles
    if data is None or len(data) < 50:
        return no_trade(
            "At least 50 candles are required."
        )

    row = data.iloc[-1]

    trend_result = analyze_trend(row)
    market_result = analyze_market_conditions(row)

    reasons = [
        trend_result["reason"],
        market_result["reason"],
    ]

    def filtered_no_trade(reason):
        return {
            "signal": "NO TRADE",
            "trend": trend_result["trend"],
            "momentum": market_result["momentum"],
            "volatility": market_result["volatility"],
            "atr_percent": market_result["atr_percent"],
            "expiry_minutes": expiry_minutes,
            "reasons": reasons + [reason],
        }

    if trend_result["trend"] == "UNKNOWN":
        return filtered_no_trade(
            "Trend could not be determined."
        )

    if not market_result["valid"]:
        return filtered_no_trade(
            market_result["reason"]
        )

    trend = trend_result["trend"]
    momentum = market_result["momentum"]

    if trend == "SIDEWAYS":
        return filtered_no_trade(
            "Sideways market: no clear trend."
        )

    if trend != momentum:
        return filtered_no_trade(
            "Trend and momentum are not aligned."
        )

    rsi = float(row["RSI_14"])

    if trend == "BULLISH" and rsi >= 70:
        return filtered_no_trade(
            "RSI is in the overbought zone."
        )

    if trend == "BEARISH" and rsi <= 30:
        return filtered_no_trade(
            "RSI is in the oversold zone."
        )

    # Check conflicting candlestick patterns.
    pattern_direction = str(
        row.get("Pattern_Direction", "NONE")
    ).upper()

    directions = {
        item.strip()
        for item in pattern_direction.split(";")
    }

    if "BULLISH" in directions and "BEARISH" in directions:
        return filtered_no_trade(
            "Conflicting candlestick patterns detected."
        )

    if trend == "BULLISH" and "BEARISH" in directions:
        return filtered_no_trade(
            "Bearish candlestick pattern conflicts with trend."
        )

    if trend == "BEARISH" and "BULLISH" in directions:
        return filtered_no_trade(
            "Bullish candlestick pattern conflicts with trend."
        )

    # Avoid signals immediately next to opposing S/R levels.
    atr = float(row["ATR_14"])
    close = float(row["Close"])

    for column in ("Support", "Resistance"):
        try:
            level = float(row[column])
        except (KeyError, TypeError, ValueError):
            level = None

        if level is not None:
            if trend == "BULLISH" and column == "Resistance":
                if abs(level - close) <= 0.25 * atr:
                    return filtered_no_trade(
                        "Price is too close to resistance."
                    )

            if trend == "BEARISH" and column == "Support":
                if abs(close - level) <= 0.25 * atr:
                    return filtered_no_trade(
                        "Price is too close to support."
                    )

    signal = "UP" if trend == "BULLISH" else "DOWN"

    reasons.append(
        "Trend, momentum and pattern checks are aligned."
    )

    if mode == "delayed":
        reasons.append(
            "WARNING: Analysis uses delayed market data."
        )

    reasons.append(
        "This is an analysis result, not a guaranteed prediction."
    )

    return {
        "signal": signal,
        "trend": trend,
        "momentum": momentum,
        "volatility": market_result["volatility"],
        "atr_percent": market_result["atr_percent"],
        "expiry_minutes": expiry_minutes,
        "reasons": reasons,
    }

"""Stage 7.3: conservative, research-only polling scheduler."""

from __future__ import annotations

import fcntl
import json
import os
from datetime import datetime, timedelta, timezone
from pathlib import Path

from market_data.twelve_data_client import (
    BudgetExceeded,
    MarketDataError,
    fetch_recent_candles,
)

POLLING_MINUTES = (1, 3, 5)
DEFAULT_POLLING_MINUTES = 3

# Global local-machine spacing, regardless of symbol or browser session.
MIN_REQUEST_SPACING_SECONDS = 60
ERROR_BACKOFF_SECONDS = 180

SCHEDULER_PATH = Path(
    "data/api_usage/live_monitor_scheduler.json"
)


def utc_now():
    return datetime.now(timezone.utc)


def _iso(dt):
    return dt.astimezone(timezone.utc).isoformat()


def _parse_time(value):
    if not value:
        return None

    dt = datetime.fromisoformat(value)
    if dt.tzinfo is None:
        raise MarketDataError("Scheduler timestamp lacks timezone")

    return dt.astimezone(timezone.utc)


def _read_state(file):
    file.seek(0)
    raw = file.read()

    if not raw.strip():
        return {}

    try:
        state = json.loads(raw)
    except json.JSONDecodeError as exc:
        raise MarketDataError(
            "Scheduler state corrupted; polling blocked"
        ) from exc

    if not isinstance(state, dict):
        raise MarketDataError("Invalid scheduler state")

    return state


def _save_state(file, state):
    file.seek(0)
    file.truncate()
    json.dump(state, file, indent=2)
    file.flush()
    os.fsync(file.fileno())


def reserve_poll(
    *,
    interval_minutes=DEFAULT_POLLING_MINUTES,
    path=SCHEDULER_PATH,
    now=None,
):
    """Reserve one poll slot across local processes.

    Uses a persistent file lock and fails closed on invalid state.
    No network request happens here.
    """
    if (
        type(interval_minutes) is not int
        or interval_minutes not in POLLING_MINUTES
    ):
        raise ValueError("Polling must be 1, 3 or 5 minutes")

    now = now or utc_now()
    if now.tzinfo is None:
        raise ValueError("Polling time must have a timezone")

    now = now.astimezone(timezone.utc)
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a+", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_EX)

        try:
            state = _read_state(file)

            last_attempt = _parse_time(
                state.get("last_attempt_utc")
            )
            next_allowed = _parse_time(
                state.get("next_allowed_utc")
            )

            if last_attempt and now < last_attempt:
                raise MarketDataError(
                    "System clock moved backwards; polling blocked"
                )

            if next_allowed and now < next_allowed:
                return {
                    "allowed": False,
                    "next_allowed_utc": _iso(next_allowed),
                    "reason": "Waiting for polling interval",
                }

            spacing = max(
                interval_minutes * 60,
                MIN_REQUEST_SPACING_SECONDS,
            )

            next_allowed = now + timedelta(
                seconds=spacing
            )

            state.update({
                "last_attempt_utc": _iso(now),
                "next_allowed_utc": _iso(next_allowed),
                "last_status": "RESERVED",
            })

            _save_state(file, state)

            return {
                "allowed": True,
                "next_allowed_utc": _iso(next_allowed),
                "reason": "Poll reserved",
            }
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def record_poll_result(
    *,
    status,
    path=SCHEDULER_PATH,
    now=None,
):
    """Record outcome without shortening an existing cooldown."""

    if status not in ("SUCCESS", "ERROR", "BUDGET_BLOCKED"):
        raise ValueError("Invalid poll status")

    now = now or utc_now()
    if now.tzinfo is None:
        raise ValueError("Result time must have a timezone")

    now = now.astimezone(timezone.utc)
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a+", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_EX)

        try:
            state = _read_state(file)

            if status != "SUCCESS":
                candidate = now + timedelta(
                    seconds=ERROR_BACKOFF_SECONDS
                )

                current = _parse_time(
                    state.get("next_allowed_utc")
                )

                if current is None or candidate > current:
                    state["next_allowed_utc"] = _iso(candidate)

            state["last_status"] = status
            state["last_result_utc"] = _iso(now)

            _save_state(file, state)
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def read_monitor_state(path=SCHEDULER_PATH):
    """Read local scheduler status safely."""
    path = Path(path)

    if not path.exists():
        return {}

    with path.open("r", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_SH)

        try:
            return _read_state(file)
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def run_monitor_poll(
    config,
    *,
    interval_minutes=DEFAULT_POLLING_MINUTES,
    scheduler_path=SCHEDULER_PATH,
    fetcher=None,
    now=None,
):
    """Attempt one scheduled fetch.

    One local-machine request slot is shared by all configurations.
    The underlying client still enforces its own API budget.
    """
    now = now or utc_now()

    slot = reserve_poll(
        interval_minutes=interval_minutes,
        path=scheduler_path,
        now=now,
    )

    if not slot["allowed"]:
        return {
            "status": "WAITING",
            "reason": slot["reason"],
            "next_allowed_utc": slot["next_allowed_utc"],
            "result": None,
        }

    fetch = fetcher or fetch_recent_candles

    try:
        result = fetch(config)

    except BudgetExceeded:
        record_poll_result(
            status="BUDGET_BLOCKED",
            path=scheduler_path,
            now=now,
        )
        return {
            "status": "BUDGET_BLOCKED",
            "reason": "Local API budget exhausted",
            "result": None,
        }

    except MarketDataError as exc:
        record_poll_result(
            status="ERROR",
            path=scheduler_path,
            now=now,
        )
        return {
            "status": "ERROR",
            "reason": str(exc),
            "result": None,
        }

    except Exception:
        record_poll_result(
            status="ERROR",
            path=scheduler_path,
            now=now,
        )
        raise

    record_poll_result(
        status="SUCCESS",
        path=scheduler_path,
        now=now,
    )

    return {
        "status": "SUCCESS",
        "reason": "Market data fetched",
        "result": result,
    }

def get_credit_dashboard_status(
    *,
    path=DEFAULT_PATH,
    now=None,
    required_credits=1,
    reserve_credits=20,
):
    now = now or datetime.now(timezone.utc)

    snapshot = load_credit_state(
        path=path,

        snapshot.credits_left
        >= required_credits + reserve_credits
    )

    return {
        "status": "READY" if allowed else "LOW_CREDITS",
        "credits_left": snapshot.credits_left,
        "credits_used": snapshot.credits_used,
        "request_allowed": allowed,
        "message": (
            "Provider credit threshold satisfied"
            if allowed
            else "Insufficient provider credits for guarded request"
        ),
    }
