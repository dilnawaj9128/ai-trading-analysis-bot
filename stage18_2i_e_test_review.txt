========== tests/test_credit_refresh_approval.py ==========
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone

import pytest

from market_data.credit_refresh_approval import (
    claim_credit_refresh_approval,
    CreditRefreshApprovalRequired,
    CreditRefreshApprovalAlreadyUsed,
)


def claim(path, approved=True, approval_id="test-approval"):
    return claim_credit_refresh_approval(
        approval_id=approval_id,
        explicitly_approved=approved,
        path=path,
        now=datetime(2026, 11, 18, tzinfo=timezone.utc),
    )


def test_explicit_approval_required(tmp_path):
    path = tmp_path / "claim.json"

    with pytest.raises(CreditRefreshApprovalRequired):
        claim(path, approved=False)

    assert not path.exists()


def test_single_use_claim(tmp_path):
    path = tmp_path / "claim.json"

    result = claim(path)
    assert result["status"] == "CLAIMED_OFFLINE"

    with pytest.raises(CreditRefreshApprovalAlreadyUsed):
        claim(path)


def test_claim_persists(tmp_path):
    path = tmp_path / "claim.json"

    claim(path)
    assert path.exists()


def test_invalid_approval_id(tmp_path):
    with pytest.raises(ValueError):
        claim(tmp_path / "claim.json", approval_id="")


def test_invalid_approval_flag(tmp_path):
    with pytest.raises(ValueError):
        claim_credit_refresh_approval(
            approval_id="test",
            explicitly_approved=1,
            path=tmp_path / "claim.json",
        )


def test_concurrent_claim_only_one_succeeds(tmp_path):
    path = tmp_path / "claim.json"

    def attempt(_):
        try:
            claim(path)
            return "SUCCESS"
        except CreditRefreshApprovalAlreadyUsed:
            return "BLOCKED"

    with ThreadPoolExecutor(max_workers=8) as executor:
        results = list(executor.map(attempt, range(8)))

    assert results.count("SUCCESS") == 1
    assert results.count("BLOCKED") == 7
========== tests/test_credit_refresh_lifecycle.py ==========
from datetime import datetime, timedelta, timezone

import pytest

from market_data.credit_refresh_lifecycle import (
    evaluate_refresh_lifecycle,
)


NOW = datetime(2026, 11, 18, 12, 0, tzinfo=timezone.utc)


def evaluate(status=None, elapsed=None, **kwargs):
    attempt = (
        NOW - timedelta(seconds=elapsed)
        if elapsed is not None else None
    )

    return evaluate_refresh_lifecycle(
        last_attempt_at=attempt,
        last_status=status,
        now=NOW,
        **kwargs,
    )


def test_no_history_eligible():
    assert evaluate().action == "ELIGIBLE"


def test_claimed_blocks():
    assert evaluate("CLAIMED", 7200).action == "BLOCK"


def test_uncertain_blocks():
    assert evaluate("UNCERTAIN", 7200).action == "BLOCK"


def test_completed_inside_cooldown():
    assert evaluate("COMPLETED", 100).action == "COOLDOWN"


def test_completed_after_cooldown():
    assert evaluate("COMPLETED", 3600).action == "ELIGIBLE"


def test_future_timestamp_blocks():
    assert evaluate("COMPLETED", -10).action == "BLOCK"


def test_invalid_status_rejected():
    with pytest.raises(ValueError):
        evaluate("INVALID", 7200)


def test_incomplete_history_rejected():
    with pytest.raises(ValueError):
        evaluate("COMPLETED")
========== tests/test_http_reservation_lifecycle.py ==========
from unittest.mock import patch

import pytest

from market_data.historical_collector import (
    fetch_historical_candles,
)
from market_data.twelve_data_client import MarketDataError
from config.market_config import create_research_config


@pytest.mark.parametrize(
    "failure",
    [
        MarketDataError("network error"),
        MarketDataError("HTTP 429"),
        MarketDataError("invalid JSON"),
    ],
)
def test_http_failure_marks_reservation_uncertain(failure):
    config = create_research_config(
        "quotex", "forex", "EUR/USD", "1min", 15
    )

    with (
        patch(
            "market_data.historical_collector._validate_historical_request"
        ) as validate,
        patch(
            "market_data.historical_collector.authorize_request"
        ) as authorize,
        patch(
            "market_data.historical_collector._perform_historical_http_request",
            side_effect=failure,
        ),
        patch(
            "market_data.authorization_ledger.transition_authorization"
        ) as transition,
    ):
        from datetime import datetime, timezone
        import pandas as pd

        now = pd.Timestamp(
            datetime(2026, 11, 13, 12, 0, tzinfo=timezone.utc)
        )

        validate.return_value = (
            now - pd.Timedelta(minutes=10),
            now,
            now,
            pd.Timedelta(minutes=1),
            "fake-test-key",
        )

        authorize.return_value.request_id = "test-request-123"

        with pytest.raises(MarketDataError):
            fetch_historical_candles(
                config,
                start=now - pd.Timedelta(minutes=10),
                end=now,
                api_key="fake-test-key",
                provider_credit_path="fake-provider.json",
            )

        transition.assert_called_once_with(
            provider_state_path="fake-provider.json",
            request_id="test-request-123",
            status="UNCERTAIN",
        )
