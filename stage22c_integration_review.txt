"""Research-only technical analysis of validated completed candles."""

import pandas as pd

from indicators import calculate_ema, calculate_indicators
from patterns.candlestick import detect_candlestick_patterns


def analyze_live_candles(frame: pd.DataFrame) -> dict:
    """Analyze completed OHLC candles without placing trades."""

    if not isinstance(frame, pd.DataFrame):
        raise TypeError("Expected pandas DataFrame")

    if len(frame) < 60:
        raise ValueError(
            "At least 60 completed candles are required"
        )

    # Existing validation and indicators are reused.
    enriched = calculate_indicators(frame)

    enriched["EMA_9"] = calculate_ema(
        enriched["Close"], 9
    )
    enriched["EMA_21"] = calculate_ema(
        enriched["Close"], 21
    )

    patterns = detect_candlestick_patterns(frame)

    latest = enriched.iloc[-1]

    ema9 = latest["EMA_9"]
    ema21 = latest["EMA_21"]
    rsi = latest["RSI_14"]

    observations = []

    if pd.notna(ema9) and pd.notna(ema21):
        if ema9 > ema21:
            observations.append(
                "Short-term EMA 9 is above EMA 21"
            )
        elif ema9 < ema21:
            observations.append(
                "Short-term EMA 9 is below EMA 21"
            )
        else:
            observations.append(
                "EMA 9 and EMA 21 are equal"
            )

    if pd.notna(rsi):
        if rsi > 70:
            observations.append(
                "RSI is above 70 (high momentum reading)"
            )
        elif rsi < 30:
            observations.append(
                "RSI is below 30 (low momentum reading)"
            )
        else:
            observations.append(
                "RSI is between 30 and 70"
            )

    return {
        "frame": enriched,
        "patterns": patterns,
        "observations": observations,
        "latest": latest.to_dict(),
        "analysis_only": True,
        "execution_enabled": False,
        "directional_backtesting_allowed": False,
    }

"""Stage 7.3: conservative, research-only polling scheduler."""

from __future__ import annotations

import fcntl
import json
import os
from datetime import datetime, timedelta, timezone
from pathlib import Path

from market_data.twelve_data_client import (
    BudgetExceeded,
    MarketDataError,
    fetch_recent_candles,
)

POLLING_MINUTES = (1, 3, 5)
DEFAULT_POLLING_MINUTES = 3

# Global local-machine spacing, regardless of symbol or browser session.
MIN_REQUEST_SPACING_SECONDS = 60
ERROR_BACKOFF_SECONDS = 180

SCHEDULER_PATH = Path(
    "data/api_usage/live_monitor_scheduler.json"
)


def utc_now():
    return datetime.now(timezone.utc)


def _iso(dt):
    return dt.astimezone(timezone.utc).isoformat()


def _parse_time(value):
    if not value:
        return None

    dt = datetime.fromisoformat(value)
    if dt.tzinfo is None:
        raise MarketDataError("Scheduler timestamp lacks timezone")

    return dt.astimezone(timezone.utc)


def _read_state(file):
    file.seek(0)
    raw = file.read()

    if not raw.strip():
        return {}

    try:
        state = json.loads(raw)
    except json.JSONDecodeError as exc:
        raise MarketDataError(
            "Scheduler state corrupted; polling blocked"
        ) from exc

    if not isinstance(state, dict):
        raise MarketDataError("Invalid scheduler state")

    return state


def _save_state(file, state):
    file.seek(0)
    file.truncate()
    json.dump(state, file, indent=2)
    file.flush()
    os.fsync(file.fileno())


def reserve_poll(
    *,
    interval_minutes=DEFAULT_POLLING_MINUTES,
    path=SCHEDULER_PATH,
    now=None,
):
    """Reserve one poll slot across local processes.

    Uses a persistent file lock and fails closed on invalid state.
    No network request happens here.
    """
    if (
        type(interval_minutes) is not int
        or interval_minutes not in POLLING_MINUTES
    ):
        raise ValueError("Polling must be 1, 3 or 5 minutes")

    now = now or utc_now()
    if now.tzinfo is None:
        raise ValueError("Polling time must have a timezone")

    now = now.astimezone(timezone.utc)
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a+", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_EX)

        try:
            state = _read_state(file)

            last_attempt = _parse_time(
                state.get("last_attempt_utc")
            )
            next_allowed = _parse_time(
                state.get("next_allowed_utc")
            )

            if last_attempt and now < last_attempt:
                raise MarketDataError(
                    "System clock moved backwards; polling blocked"
                )

            if next_allowed and now < next_allowed:
                return {
                    "allowed": False,
                    "next_allowed_utc": _iso(next_allowed),
                    "reason": "Waiting for polling interval",
                }

            spacing = max(
                interval_minutes * 60,
                MIN_REQUEST_SPACING_SECONDS,
            )

            next_allowed = now + timedelta(
                seconds=spacing
            )

            state.update({
                "last_attempt_utc": _iso(now),
                "next_allowed_utc": _iso(next_allowed),
                "last_status": "RESERVED",
            })

            _save_state(file, state)

            return {
                "allowed": True,
                "next_allowed_utc": _iso(next_allowed),
                "reason": "Poll reserved",
            }
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def record_poll_result(
    *,
    status,
    path=SCHEDULER_PATH,
    now=None,
):
    """Record outcome without shortening an existing cooldown."""

    if status not in ("SUCCESS", "ERROR", "BUDGET_BLOCKED"):
        raise ValueError("Invalid poll status")

    now = now or utc_now()
    if now.tzinfo is None:
        raise ValueError("Result time must have a timezone")

    now = now.astimezone(timezone.utc)
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a+", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_EX)

        try:
            state = _read_state(file)

            if status != "SUCCESS":
                candidate = now + timedelta(
                    seconds=ERROR_BACKOFF_SECONDS
                )

                current = _parse_time(
                    state.get("next_allowed_utc")
                )

                if current is None or candidate > current:
                    state["next_allowed_utc"] = _iso(candidate)

            state["last_status"] = status
            state["last_result_utc"] = _iso(now)

            _save_state(file, state)
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def read_monitor_state(path=SCHEDULER_PATH):
    """Read local scheduler status safely."""
    path = Path(path)

    if not path.exists():
        return {}

    with path.open("r", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_SH)

        try:
            return _read_state(file)
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def run_monitor_poll(
    config,
    *,
    interval_minutes=DEFAULT_POLLING_MINUTES,
    scheduler_path=SCHEDULER_PATH,
    fetcher=None,
    now=None,
):
    """Attempt one scheduled fetch.

    One local-machine request slot is shared by all configurations.
    The underlying client still enforces its own API budget.
    """
    now = now or utc_now()

    slot = reserve_poll(
        interval_minutes=interval_minutes,
        path=scheduler_path,
        now=now,
    )

    if not slot["allowed"]:
        return {
            "status": "WAITING",
            "reason": slot["reason"],
            "next_allowed_utc": slot["next_allowed_utc"],
            "result": None,
        }

    fetch = fetcher or fetch_recent_candles

    try:
        result = fetch(config)

    except BudgetExceeded:
        record_poll_result(
            status="BUDGET_BLOCKED",
            path=scheduler_path,
            now=now,
        )
        return {
            "status": "BUDGET_BLOCKED",
            "reason": "Local API budget exhausted",
            "result": None,
        }

    except MarketDataError as exc:
        record_poll_result(
            status="ERROR",
            path=scheduler_path,
            now=now,
        )
        return {
            "status": "ERROR",
            "reason": str(exc),
            "result": None,
        }

    except Exception:
        record_poll_result(
            status="ERROR",
            path=scheduler_path,
            now=now,
        )
        raise

    record_poll_result(
        status="SUCCESS",
        path=scheduler_path,
        now=now,
    )

    return {
        "status": "SUCCESS",
        "reason": "Market data fetched",
        "result": result,
    }

"""Local analysis notifications. No network or trade execution."""

import sqlite3
from datetime import datetime, timezone
from pathlib import Path

DEFAULT_DB = Path("data/notifications/analysis_notifications.db")
SIGNALS = {"UP", "DOWN", "NO TRADE"}


def record_analysis_notification(
    *,
    symbol,
    timeframe,
    candle_timestamp,
    signal,
    expiry_minutes=15,
    enabled=True,
    db_path=DEFAULT_DB,
):
    """Store one notification per completed candle and expiry."""

    if type(enabled) is not bool:
        raise ValueError("enabled must be boolean")

    if not enabled:
        return {"created": False, "reason": "DISABLED"}

    if signal not in SIGNALS:
        raise ValueError("Invalid signal")

    if not isinstance(symbol, str) or not symbol.strip():
        raise ValueError("Invalid symbol")

    if timeframe not in {"1min", "5min", "15min"}:
        raise ValueError("Invalid timeframe")

    if type(expiry_minutes) is not int or expiry_minutes not in (1, 5, 15):
        raise ValueError("Invalid expiry")

    stamp = datetime.fromisoformat(
        str(candle_timestamp).replace("Z", "+00:00")
    )

    if stamp.tzinfo is None:
        raise ValueError("Timezone-aware timestamp required")

    stamp = stamp.astimezone(timezone.utc).isoformat()

    path = Path(db_path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with sqlite3.connect(path, timeout=10) as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS analysis_notifications (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                symbol TEXT NOT NULL,
                timeframe TEXT NOT NULL,
                candle_timestamp TEXT NOT NULL,
                signal TEXT NOT NULL,
                expiry_minutes INTEGER NOT NULL,
                created_at TEXT NOT NULL,
                UNIQUE (
                    symbol,
                    timeframe,
                    candle_timestamp,
                    expiry_minutes
                )
            )
        """)

        cursor = conn.execute("""
            INSERT OR IGNORE INTO analysis_notifications (
                symbol, timeframe, candle_timestamp,
                signal, expiry_minutes, created_at
            ) VALUES (?, ?, ?, ?, ?, ?)
        """, (
            symbol.strip(),
            timeframe,
            stamp,
            signal,
            expiry_minutes,
            datetime.now(timezone.utc).isoformat(),
        ))

        created = cursor.rowcount == 1

    return {
        "created": created,
        "reason": "NEW_RESULT" if created else "DUPLICATE",
        "signal": signal,
        "analysis_only": True,
        "execution_enabled": False,
    }


def get_notification_history(
    *,
    db_path=DEFAULT_DB,
    limit=50,
):
    if type(limit) is not int or not 1 <= limit <= 500:
        raise ValueError("Invalid history limit")

    path = Path(db_path)

    if not path.exists():
        return []

    with sqlite3.connect(path) as conn:
        conn.row_factory = sqlite3.Row

        rows = conn.execute("""
            SELECT symbol, timeframe, candle_timestamp,
                   signal, expiry_minutes, created_at
            FROM analysis_notifications
            ORDER BY id DESC
            LIMIT ?
        """, (limit,)).fetchall()

    return [dict(row) for row in rows]

"""Local notification preferences. No network requests."""

import json
import os
import tempfile
from pathlib import Path

DEFAULT_PATH = Path("data/notifications/preferences.json")


def load_notification_enabled(path=DEFAULT_PATH):
    path = Path(path)

    if not path.exists():
        return False

    data = json.loads(path.read_text(encoding="utf-8"))

    if (
        type(data) is not dict
        or data.get("version") != 1
        or type(data.get("enabled")) is not bool
    ):
        raise ValueError("Invalid notification preferences")

    return data["enabled"]


def save_notification_enabled(enabled, path=DEFAULT_PATH):
    if type(enabled) is not bool:
        raise ValueError("Notification setting must be boolean")

    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    temporary = None

    try:
        with tempfile.NamedTemporaryFile(
            mode="w",
            encoding="utf-8",
            dir=path.parent,
            prefix=".notification_preferences_",
            suffix=".tmp",
            delete=False,
        ) as handle:
            temporary = Path(handle.name)

            json.dump(
                {"version": 1, "enabled": enabled},
                handle,
            )

            handle.flush()
            os.fsync(handle.fileno())

        os.replace(temporary, path)

    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)

    return enabled
