"""Local read-only historical research dashboard."""

from pathlib import Path

import pandas as pd
import streamlit as st

from dashboard.report_loader import (
    DashboardReportError,
    dashboard_summary,
    hourly_chart_data,
    load_report,
    volatility_chart_data,
)


DEFAULT_REPORT = (
    "reports/eurusd_advanced_20261112.json"
)


def main():
    st.set_page_config(
        page_title="AI Trading Research Dashboard",
        page_icon="📊",
        layout="wide",
    )

    st.title("AI Trading Research Dashboard")
    st.caption(
        "Historical EUR/USD research | UTC | Read-only"
    )

    st.warning(
        "ANALYSIS ONLY — No trading signals, "
        "broker execution, or directional backtesting. "
        "Feed provenance remains unverified."
    )

    with st.sidebar:
        st.header("Research report")

        report_path = st.text_input(
            "Local JSON report path",
            value=DEFAULT_REPORT,
        )

        st.info(
            "This dashboard only reads an existing "
            "schema v3 research report."
        )

    try:
        report = load_report(Path(report_path))
        summary = dashboard_summary(report)
    except DashboardReportError as exc:
        st.error(f"Report validation failed: {exc}")
        st.stop()

    st.subheader("Dataset overview")

    columns = st.columns(4)

    columns[0].metric(
        "Accepted sessions",
        summary["accepted_sessions"],
    )
    columns[1].metric(
        "Unique candles",
        summary["unique_candles"],
    )
    columns[2].metric(
        "Duplicates",
        summary["duplicates"],
    )
    columns[3].metric(
        "Overlaps",
        summary["overlaps"],
    )

    coverage = report[
        "advanced_analytics"
    ]["coverage"]

    st.subheader("Historical data coverage")

    left, right = st.columns(2)

    with left:
        percentage = summary["coverage_percent"]

        st.metric(
            "Observed-window coverage",
            (
                f"{percentage:.2f}%"
                if percentage is not None
                else "N/A"
            ),
        )

        st.metric(
            "Missing minutes",
            summary["missing_minutes"],
        )

    with right:
        st.write(
            "First candle:",
            coverage["first_timestamp"],
        )
        st.write(
            "Last candle:",
            coverage["last_timestamp"],
        )

        st.caption(
            "Missing minutes are calculated between "
            "the first and last accepted timestamps. "
            "Market closures are not excluded."
        )

    st.subheader("Hourly candle behavior")

    hourly = pd.DataFrame(
        hourly_chart_data(report)
    )

    if hourly.empty:
        st.info("No accepted hourly data.")
    else:
        st.bar_chart(
            hourly.set_index("hour")[
                ["average_range"]
            ]
        )

        st.dataframe(
            hourly,
            use_container_width=True,
            hide_index=True,
        )

    st.subheader("Historical volatility ranking")

    ranking = pd.DataFrame(
        volatility_chart_data(report)
    )

    if ranking.empty:
        st.info("No accepted sessions to rank.")
    else:
        st.bar_chart(
            ranking.set_index("session")[
                ["average_range"]
            ]
        )

        st.dataframe(
            ranking,
            use_container_width=True,
            hide_index=True,
        )

    st.subheader("Session quality and selection")

    sessions = report.get("sessions", [])

    if sessions:
        session_rows = []

        for item in sessions:
            session_rows.append({
                "Dataset": item.get("dataset"),
                "Quality": item.get("quality", {}).get(
                    "status", "UNKNOWN"
                ) if isinstance(
                    item.get("quality"), dict
                ) else "See report",
                "Selection": item.get(
                    "selection_status"
                ),
                "Reason": item.get(
                    "selection_reason"
                ),
            })

        st.dataframe(
            pd.DataFrame(session_rows),
            use_container_width=True,
            hide_index=True,
        )
    else:
        st.info("No session records.")

    with st.expander("Research limitations"):
        for limitation in report.get(
            "limitations", []
        ):
            st.write("-", limitation)

    st.divider()

    st.caption(
        "Research-only visualization. "
        "Historical volatility does not predict "
        "future price direction."
    )


if __name__ == "__main__":
    main()

from pathlib import Path

COLUMNS = ("timestamp", "Open", "High", "Low", "Close")


def accepted_datasets(report):
    """Only use datasets accepted by the session selector."""

    sessions = report.get("sessions", [])

    if not isinstance(sessions, list):
        raise DashboardReportError("Invalid session records")

    selected = []

    for session in sessions:
        if not isinstance(session, dict):
            raise DashboardReportError("Invalid session entry")

        if session.get("selection_status") == "ACCEPTED":
            directory = session.get("dataset")

            if not isinstance(directory, str) or not directory:
                raise DashboardReportError(
                    "Accepted session has no dataset path"
                )

            selected.append(directory)

    if len(selected) != report["datasets_accepted"]:
        raise DashboardReportError(
            "Accepted dataset count mismatch"
        )

    return selected


def load_chart_candles(dataset_dir):
    """
    Re-audit original CSV and receipt before charting.

    No file modifications, no broker access.
    """

    directory = Path(dataset_dir)

    csv_path = directory / CSV_NAME
    receipt_path = directory / RECEIPT_NAME

    audit = audit_dataset(csv_path, receipt_path)

    if audit.get("status") != "PASS":
        raise DashboardReportError(
            "Dataset failed quality re-audit"
        )

    frame = pd.read_csv(csv_path)

    if not set(COLUMNS).issubset(frame.columns):
        raise DashboardReportError(
            "Missing OHLC or timestamp columns"
        )

    frame = frame.loc[:, COLUMNS].copy()

    frame["timestamp"] = pd.to_datetime(
        frame["timestamp"],
        utc=True,
        errors="raise",
    )

    frame = frame.sort_values(
        "timestamp"
    ).reset_index(drop=True)

    return frame


def calculate_indicators(frame):
    """
    Historical EMA 9/21 and Wilder RSI 14.

    Warmup values remain unavailable (NaN).
    No directional classification is produced.
    """

    result = frame.copy()

    closes = result["Close"].astype(float)

    result["EMA_9"] = closes.ewm(
        span=9,
        adjust=False,
        min_periods=9,
    ).mean()

    result["EMA_21"] = closes.ewm(
        span=21,
        adjust=False,
        min_periods=21,
    ).mean()

    # Wilder RSI: initial 14-period mean, then
    # recursive smoothing of gains and losses.
    period = 14
    delta = closes.diff()
    gains = delta.clip(lower=0)
    losses = (-delta).clip(lower=0)

    rsi = pd.Series(
        float("nan"),
        index=result.index,
        dtype=float,
    )

    if len(result) > period:
        avg_gain = gains.iloc[1:period + 1].mean()
        avg_loss = losses.iloc[1:period + 1].mean()

        for index in range(period, len(result)):
            if index > period:
                avg_gain = (
                    avg_gain * (period - 1)
                    + float(gains.iloc[index])
                ) / period

                avg_loss = (
                    avg_loss * (period - 1)
                    + float(losses.iloc[index])
                ) / period

            if avg_loss == 0 and avg_gain == 0:
                value = 50.0
            elif avg_loss == 0:
                value = 100.0
            elif avg_gain == 0:
                value = 0.0
            else:
                relative_strength = avg_gain / avg_loss
                value = 100 - (
                    100 / (1 + relative_strength)
                )

            rsi.iloc[index] = value

    result["RSI_14"] = rsi

    return result


def filter_time_window(frame, start, end):
    """Inclusive UTC timestamp filter."""

    start_utc = pd.Timestamp(start)
    end_utc = pd.Timestamp(end)

    if start_utc.tzinfo is None:
        start_utc = start_utc.tz_localize("UTC")
    else:
        start_utc = start_utc.tz_convert("UTC")

    if end_utc.tzinfo is None:
        end_utc = end_utc.tz_localize("UTC")
    else:
        end_utc = end_utc.tz_convert("UTC")

    if start_utc > end_utc:
        raise ValueError("Start time must precede end time")

    return frame.loc[
        frame["timestamp"].between(
            start_utc, end_utc
        )
    ].copy()


def session_comparison(report):
    """Build descriptive comparisons from accepted sessions."""

    datasets = accepted_datasets(report)
    rows = []

    for index, directory in enumerate(
        datasets, start=1
    ):
        frame = load_chart_candles(directory)

        ranges = (
            frame["High"].astype(float)
            - frame["Low"].astype(float)
        )

        rows.append({
            "Session": index,
            "Dataset": directory,
            "Candles": len(frame),
            "Average range": float(ranges.mean()),
            "Maximum range": float(ranges.max()),
            "First UTC": frame["timestamp"].iloc[0],
            "Last UTC": frame["timestamp"].iloc[-1],
        })

    return pd.DataFrame(rows)


def load_chart_report(report_path):
    """Reuse the existing safety-validated JSON loader."""

    return load_report(report_path)

def fmt(value, digits=5):
    try:
        return "N/A" if value is None or pd.isna(value) else f"{float(value):.{digits}f}"
    except (ValueError, TypeError):
        return "N/A"


@st.fragment(run_every="10s")
def live_panel():
    if not st.session_state.get("simple_running"):
        return
    try:
        outcome = run_monitor_poll(config, interval_minutes=refresh_minutes)
    except (MarketDataError, ValueError, OSError) as exc:
        st.error(f"Monitoring unavailable: {exc}")
        return

    status = outcome["status"]
    if status == "SUCCESS":
        st.session_state["simple_result"] = outcome["result"]
        st.session_state["simple_last_fetch"] = datetime.now(timezone.utc).isoformat()
    elif status == "WAITING":
        st.caption("Next permitted fetch: " + outcome["next_allowed_utc"])
    elif status == "BUDGET_BLOCKED":
        st.error("Local API request budget reached. Fetching paused.")
    else:
        st.error("Fetch failed: " + outcome.get("reason", "Unknown error"))

        return

        timestamp = pd.Timestamp(result["latest_utc"])

========== analysis/live_engine.py ==========

"""Research-only technical analysis of validated completed candles."""

import pandas as pd

from indicators import calculate_ema, calculate_indicators
from patterns.candlestick import detect_candlestick_patterns


def analyze_live_candles(frame: pd.DataFrame) -> dict:
    """Analyze completed OHLC candles without placing trades."""

    if not isinstance(frame, pd.DataFrame):
        raise TypeError("Expected pandas DataFrame")

    if len(frame) < 60:
        raise ValueError(
            "At least 60 completed candles are required"
        )

    # Existing validation and indicators are reused.
    enriched = calculate_indicators(frame)

    enriched["EMA_9"] = calculate_ema(
        enriched["Close"], 9
    )
    enriched["EMA_21"] = calculate_ema(
        enriched["Close"], 21
    )

    patterns = detect_candlestick_patterns(frame)

    latest = enriched.iloc[-1]

    ema9 = latest["EMA_9"]
    ema21 = latest["EMA_21"]
    rsi = latest["RSI_14"]

    observations = []

    if pd.notna(ema9) and pd.notna(ema21):
        if ema9 > ema21:
            observations.append(
                "Short-term EMA 9 is above EMA 21"
            )
        elif ema9 < ema21:
            observations.append(
                "Short-term EMA 9 is below EMA 21"
            )
        else:
            observations.append(
                "EMA 9 and EMA 21 are equal"
            )

    if pd.notna(rsi):
        if rsi > 70:
            observations.append(
                "RSI is above 70 (high momentum reading)"
            )
        elif rsi < 30:
            observations.append(
                "RSI is below 30 (low momentum reading)"
            )
        else:
            observations.append(
                "RSI is between 30 and 70"
            )

    return {
        "frame": enriched,
        "patterns": patterns,
        "observations": observations,
        "latest": latest.to_dict(),
        "analysis_only": True,
        "execution_enabled": False,
        "directional_backtesting_allowed": False,
    }

from dataclasses import asdict, dataclass
from typing import Final


class ResearchConfigError(ValueError):
    """Invalid or unsupported research configuration."""


PLATFORMS: Final = {
    "quotex": {
        "label": "Quotex",
        "mode": "external_research",
    },
    "olymp_trade": {
        "label": "Olymp Trade",
        "mode": "external_research",
    },
}

MARKETS: Final = {
    "forex": ("EUR/USD", "GBP/USD", "USD/JPY"),
    "crypto": ("BTC/USD", "ETH/USD"),
    "stocks": ("AAPL", "TSLA", "NVDA"),
    "etfs": ("SPY", "QQQ"),
    "commodities": ("XAU/USD", "XAG/USD"),
}

TIMEFRAMES: Final = {
    "1min": 1,
    "5min": 5,
    "15min": 15,
}

EXPIRY_MINUTES: Final = (15,)


@dataclass(frozen=True)
class ResearchConfig:
    platform: str
    market: str
    symbol: str
    timeframe: str
    expiry_minutes: int

    provider: str = "twelve_data"
    analysis_only: bool = True
    execution_enabled: bool = False
    broker_feed_verified: bool = False
    provider_feed_verified: bool = False
    directional_backtesting_allowed: bool = False

    def to_dict(self):
        return asdict(self)


def create_research_config(
    platform: str,
    market: str,
    symbol: str,
    timeframe: str = "1min",
    expiry_minutes: int = 15,
) -> ResearchConfig:
    """
    Validate a research profile.

    No credentials, orders, signals or broker connectivity.
    """

    if platform not in PLATFORMS:
        raise ResearchConfigError(
            f"Unsupported platform: {platform}"
        )

    if market not in MARKETS:
        raise ResearchConfigError(
            f"Unsupported market: {market}"
        )

    if symbol not in MARKETS[market]:
        raise ResearchConfigError(
            f"{symbol} is not configured for {market}"
        )

    if timeframe not in TIMEFRAMES:
        raise ResearchConfigError(
            f"Unsupported timeframe: {timeframe}"
        )

    if (
        type(expiry_minutes) is not int
        or expiry_minutes not in EXPIRY_MINUTES
    ):
        raise ResearchConfigError(
            "Only 15-minute research expiry is supported"
        )

    return ResearchConfig(
        platform=platform,
        market=market,
        symbol=symbol,
        timeframe=timeframe,
        expiry_minutes=expiry_minutes,
    )


def research_disclaimer(config: ResearchConfig) -> str:
    label = PLATFORMS[config.platform]["label"]

    return (
        f"{label} external research only. "
        "Twelve Data and broker quotes may differ. "
        "No OTC support, verified broker feed, "
        "trade execution, directional backtesting "
        "or profitability claims."
    )
