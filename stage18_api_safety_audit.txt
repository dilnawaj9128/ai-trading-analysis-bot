"""Stage 7.3: conservative, research-only polling scheduler."""

from __future__ import annotations

import fcntl
import json
import os
from datetime import datetime, timedelta, timezone
from pathlib import Path

from market_data.twelve_data_client import (
    BudgetExceeded,
    MarketDataError,
    fetch_recent_candles,
)

POLLING_MINUTES = (1, 3, 5)
DEFAULT_POLLING_MINUTES = 3

# Global local-machine spacing, regardless of symbol or browser session.
MIN_REQUEST_SPACING_SECONDS = 60
ERROR_BACKOFF_SECONDS = 180

SCHEDULER_PATH = Path(
    "data/api_usage/live_monitor_scheduler.json"
)


def utc_now():
    return datetime.now(timezone.utc)


def _iso(dt):
    return dt.astimezone(timezone.utc).isoformat()


def _parse_time(value):
    if not value:
        return None

    dt = datetime.fromisoformat(value)
    if dt.tzinfo is None:
        raise MarketDataError("Scheduler timestamp lacks timezone")

    return dt.astimezone(timezone.utc)


def _read_state(file):
    file.seek(0)
    raw = file.read()

    if not raw.strip():
        return {}

    try:
        state = json.loads(raw)
    except json.JSONDecodeError as exc:
        raise MarketDataError(
            "Scheduler state corrupted; polling blocked"
        ) from exc

    if not isinstance(state, dict):
        raise MarketDataError("Invalid scheduler state")

    return state


def _save_state(file, state):
    file.seek(0)
    file.truncate()
    json.dump(state, file, indent=2)
    file.flush()
    os.fsync(file.fileno())


def reserve_poll(
    *,
    interval_minutes=DEFAULT_POLLING_MINUTES,
    path=SCHEDULER_PATH,
    now=None,
):
    """Reserve one poll slot across local processes.

    Uses a persistent file lock and fails closed on invalid state.
    No network request happens here.
    """
    if (
        type(interval_minutes) is not int
        or interval_minutes not in POLLING_MINUTES
    ):
        raise ValueError("Polling must be 1, 3 or 5 minutes")

    now = now or utc_now()
    if now.tzinfo is None:
        raise ValueError("Polling time must have a timezone")

    now = now.astimezone(timezone.utc)
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a+", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_EX)

        try:
            state = _read_state(file)

            last_attempt = _parse_time(
                state.get("last_attempt_utc")
            )
            next_allowed = _parse_time(
                state.get("next_allowed_utc")
            )

            if last_attempt and now < last_attempt:
                raise MarketDataError(
                    "System clock moved backwards; polling blocked"
                )

            if next_allowed and now < next_allowed:
                return {
                    "allowed": False,
                    "next_allowed_utc": _iso(next_allowed),
                    "reason": "Waiting for polling interval",
                }

            spacing = max(
                interval_minutes * 60,
                MIN_REQUEST_SPACING_SECONDS,
            )

            next_allowed = now + timedelta(
                seconds=spacing
            )

            state.update({
                "last_attempt_utc": _iso(now),
                "next_allowed_utc": _iso(next_allowed),
                "last_status": "RESERVED",
            })

            _save_state(file, state)

            return {
                "allowed": True,
                "next_allowed_utc": _iso(next_allowed),
                "reason": "Poll reserved",
            }
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def record_poll_result(
    *,
    status,
    path=SCHEDULER_PATH,
    now=None,
):
    """Record outcome without shortening an existing cooldown."""

    if status not in ("SUCCESS", "ERROR", "BUDGET_BLOCKED"):
        raise ValueError("Invalid poll status")

    now = now or utc_now()
    if now.tzinfo is None:
        raise ValueError("Result time must have a timezone")

    now = now.astimezone(timezone.utc)
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open("a+", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_EX)

        try:
            state = _read_state(file)

            if status != "SUCCESS":
                candidate = now + timedelta(
                    seconds=ERROR_BACKOFF_SECONDS
                )

                current = _parse_time(
                    state.get("next_allowed_utc")
                )

                if current is None or candidate > current:
                    state["next_allowed_utc"] = _iso(candidate)

            state["last_status"] = status
            state["last_result_utc"] = _iso(now)

            _save_state(file, state)
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def read_monitor_state(path=SCHEDULER_PATH):
    """Read local scheduler status safely."""
    path = Path(path)

    if not path.exists():
        return {}

    with path.open("r", encoding="utf-8") as file:
        fcntl.flock(file.fileno(), fcntl.LOCK_SH)

        try:
            return _read_state(file)
        finally:
            fcntl.flock(file.fileno(), fcntl.LOCK_UN)


def run_monitor_poll(
    config,
    *,
    interval_minutes=DEFAULT_POLLING_MINUTES,
    scheduler_path=SCHEDULER_PATH,
    fetcher=None,
    now=None,
):
    """Attempt one scheduled fetch.

    One local-machine request slot is shared by all configurations.
    The underlying client still enforces its own API budget.
    """
    now = now or utc_now()

    slot = reserve_poll(
        interval_minutes=interval_minutes,
        path=scheduler_path,
        now=now,
    )

    if not slot["allowed"]:
        return {
            "status": "WAITING",
            "reason": slot["reason"],
            "next_allowed_utc": slot["next_allowed_utc"],
            "result": None,
        }

    fetch = fetcher or fetch_recent_candles

    try:
        result = fetch(config)

    except BudgetExceeded:
        record_poll_result(
            status="BUDGET_BLOCKED",
            path=scheduler_path,
            now=now,
        )
        return {
            "status": "BUDGET_BLOCKED",
            "reason": "Local API budget exhausted",
            "result": None,
        }

    except MarketDataError as exc:
        record_poll_result(
            status="ERROR",
            path=scheduler_path,
            now=now,
        )
        return {
            "status": "ERROR",
            "reason": str(exc),
            "result": None,
        }

    except Exception:
        record_poll_result(
            status="ERROR",
            path=scheduler_path,
            now=now,
        )
        raise

    record_poll_result(
        status="SUCCESS",
        path=scheduler_path,
        now=now,
    )

    return {
        "status": "SUCCESS",
        "reason": "Market data fetched",
        "result": result,
    }

# Conservative local limits, not a guarantee of account entitlements.
MINUTE_LIMIT = 8
DAY_LIMIT = 800
RESERVE_PER_REQUEST = 1
TIMEOUT_SECONDS = 15
MAX_CANDLES = 120

INTERVAL_SECONDS = {
    "1min": 60,
    "5min": 300,
    "15min": 900,

class MarketDataError(RuntimeError):
    pass


class BudgetExceeded(MarketDataError):
    pass


def utc_now():
    return datetime.now(timezone.utc)


def _validate_limits():
    if MINUTE_LIMIT < 1 or DAY_LIMIT < 1:
        raise MarketDataError("Invalid local API budget limits")
    if RESERVE_PER_REQUEST < 1:
        raise MarketDataError("Invalid request reservation")



def _save_budget_atomic(path, payload):
    """Atomically replace the budget state and sync its directory."""
    temporary = None

    try:
        with tempfile.NamedTemporaryFile(
            mode="w",
            encoding="utf-8",
            dir=path.parent,
            prefix=".budget_",
            suffix=".tmp",
            delete=False,
        ) as handle:
            temporary = Path(handle.name)
            json.dump(payload, handle, indent=2)
            handle.flush()
            os.fsync(handle.fileno())

        os.replace(temporary, path)

        directory_fd = os.open(path.parent, os.O_RDONLY)
        try:
            os.fsync(directory_fd)
        finally:
            os.close(directory_fd)

    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)


def reserve_request(path=STATE_PATH, now=None):
    """Reserve estimated API usage before network access.

    Uses a separate stable lock file and atomic budget-state writes.
    Reservations are never automatically refunded.
    """
    _validate_limits()

    now = now or utc_now()
    if now.tzinfo is None:
        raise MarketDataError("Budget time must be timezone-aware")

    now = now.astimezone(timezone.utc)
    minute = now.strftime("%Y-%m-%dT%H:%MZ")
    day = now.strftime("%Y-%m-%d")

    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    lock_path = path.with_name(path.name + ".lock")

    with lock_path.open("a+b") as lock_file:
        fcntl.flock(lock_file.fileno(), fcntl.LOCK_EX)

        try:
            # Fail closed if the lock path was replaced.
            locked = os.fstat(lock_file.fileno())
            current = os.stat(lock_path)

            if (
                locked.st_dev != current.st_dev
                or locked.st_ino != current.st_ino
            ):
                raise MarketDataError(
                    "Budget lock identity changed"
                )

            try:
                raw = path.read_text(encoding="utf-8")
            except FileNotFoundError:
                # Only a genuinely missing file is a fresh budget.
                state = {}
            else:
                if not raw.strip():
                    raise MarketDataError(
                        "Existing budget state empty or corrupted; "
                        "refusing API request"
                    )

                try:
                    state = json.loads(raw)
                except json.JSONDecodeError as exc:
                    raise MarketDataError(
                        "Budget state corrupted; refusing API request"
                    ) from exc

            if not isinstance(state, dict):
                raise MarketDataError("Invalid budget state")

            if state.get("minute") != minute:
                minute_used = 0
            else:
                minute_used = state.get("minute_used", 0)

            if state.get("day") != day:
                day_used = 0
            else:
                day_used = state.get("day_used", 0)

            if (
                type(minute_used) is not int
                or type(day_used) is not int
                or minute_used < 0
                or day_used < 0
            ):
                raise MarketDataError("Invalid budget counters")

            if minute_used + RESERVE_PER_REQUEST > MINUTE_LIMIT:
                raise BudgetExceeded("Local minute budget exhausted")

            if day_used + RESERVE_PER_REQUEST > DAY_LIMIT:
                raise BudgetExceeded("Local daily budget exhausted")

            updated = {
                "minute": minute,
                "minute_used": minute_used + RESERVE_PER_REQUEST,
                "day": day,
                "day_used": day_used + RESERVE_PER_REQUEST,
            }

            _save_budget_atomic(path, updated)
            return updated

        finally:
            fcntl.flock(lock_file.fileno(), fcntl.LOCK_UN)



def parse_candles(payload, config, now=None):
    """Parse and validate recent OHLC candles.

    Returned candles are research-only. In-progress candles
    are excluded conservatively using UTC interval boundaries.
    """
    now = now or utc_now()
    if now.tzinfo is None:
        raise MarketDataError("Current time must have a timezone")
    now = now.astimezone(timezone.utc)

    if not isinstance(payload, dict):
        raise MarketDataError("Invalid API response structure")

    if payload.get("status") == "error" or payload.get("code"):
        raise MarketDataError(
            f"Provider rejected request: {payload.get('message', 'unknown')}"
        )

    values = payload.get("values")
    if not isinstance(values, list) or not values:
        raise MarketDataError("API returned no candles")

    metadata = payload.get("meta")
    if not isinstance(metadata, dict):
        raise MarketDataError("Missing provider metadata")

    returned_symbol = metadata.get("symbol")
    returned_interval = metadata.get("interval")

    if returned_symbol != config.symbol:
        raise MarketDataError("Provider symbol mismatch")

    if returned_interval != config.timeframe:
        raise MarketDataError("Provider interval mismatch")

    interval = INTERVAL_SECONDS[config.timeframe]
    rows = []
    seen = set()

    for item in values:
        if not isinstance(item, dict):
            raise MarketDataError("Malformed candle")

        try:
            stamp = datetime.strptime(
                item["datetime"], "%Y-%m-%d %H:%M:%S"
            ).replace(tzinfo=timezone.utc)

            o = float(item["open"])
            h = float(item["high"])
            l = float(item["low"])
            c = float(item["close"])
        except (KeyError, ValueError, TypeError) as exc:
            raise MarketDataError("Invalid candle fields") from exc

        if not all(math.isfinite(x) for x in (o, h, l, c)):
            raise MarketDataError("Non-finite OHLC value")

        if l > min(o, c) or h < max(o, c) or l > h:
            raise MarketDataError("Invalid OHLC relationship")

        if min(o, h, l, c) <= 0:
            raise MarketDataError("Non-positive OHLC value")

        if stamp in seen:
            raise MarketDataError("Duplicate candle timestamp")
        seen.add(stamp)

        if stamp > now:
            raise MarketDataError("Future candle timestamp")

        # Exclude candles whose interval has not yet ended.
        if stamp + timedelta(seconds=interval) > now:
            continue

        rows.append({
            "timestamp": stamp,
            "Open": o,
            "High": h,
            "Low": l,
            "Close": c,
        })

    if not rows:
        raise MarketDataError("No completed candles available")

    frame = pd.DataFrame(rows).sort_values("timestamp")
    frame = frame.reset_index(drop=True)

    # Never silently treat gaps as continuous data.
    diffs = frame["timestamp"].diff().dropna()
    gaps = int((diffs > pd.Timedelta(seconds=interval)).sum())

    latest = frame["timestamp"].iloc[-1].to_pydatetime()
    age_seconds = (now - latest).total_seconds()

    # A conservative monitoring rule, not proof of feed quality.
    stale = age_seconds > interval * 3

    return {
        "frame": frame,
        "symbol": config.symbol,
        "interval": config.timeframe,
        "candles": len(frame),
        "gaps": gaps,
        "latest_utc": latest.isoformat(),
        "age_seconds": age_seconds,
        "stale": stale,
        "analysis_only": True,
        "execution_enabled": False,
        "directional_backtesting_allowed": False,
    }


def fetch_recent_candles(
    config: ResearchConfig,
    *,
    api_key=None,
    state_path=STATE_PATH,
    now=None,
    session=None,
):
    """Fetch recent candles with budget reservation first."""

    # Revalidate even if caller constructs a dataclass directly.
    validated = create_research_config(
        config.platform,
        config.market,
        config.symbol,
        config.timeframe,
        config.expiry_minutes,
    )

    if config != validated:
        raise MarketDataError("Unsafe or modified research config")

    key = (
        api_key
        if api_key is not None
        else os.environ.get("TWELVE_DATA_API_KEY")
    )
    if not key:
        raise MarketDataError("TWELVE_DATA_API_KEY is missing")

    now = now or utc_now()
    if now.tzinfo is None:
        raise MarketDataError("Current time must have a timezone")
    now = now.astimezone(timezone.utc)

    reserve_request(state_path, now=now)

    client = session if session is not None else requests

    try:
        response = client.get(
            API_URL,
            params={
                "symbol": config.symbol,
                "interval": config.timeframe,
                "outputsize": MAX_CANDLES,
                "timezone": "UTC",
                "apikey": key,
                "format": "JSON",
            },
            timeout=TIMEOUT_SECONDS,
        )
        response.raise_for_status()
        payload = response.json()
    except (requests.RequestException, ValueError) as exc:
        raise MarketDataError(
            "Twelve Data request failed or returned invalid JSON"
        ) from exc

    return parse_candles(payload, config, now=now)

def _load_ledger(path):
    if not path.exists():
        return {
            "version": 2,
            "reserved": 0,
            "legacy_reserved": 0,
            "requests": {},
        }

    try:
        payload = json.loads(path.read_text())

        if type(payload) is not dict:
            raise ValueError("Invalid ledger")

        version = payload.get("version")

        if version == 1:
            # Legacy V1 has aggregate reservations only.
            # Reject unknown fields instead of silently
            # discarding potentially important evidence.
            if set(payload) != {"version", "reserved"}:
                raise ValueError(
                    "Invalid legacy ledger structure"
                )

            reserved = payload.get("reserved")

            if type(reserved) is not int or reserved < 0:
                raise ValueError("Invalid legacy reservation")

            return {
                "version": 2,
                "reserved": reserved,
                "legacy_reserved": reserved,
                "requests": {},
            }

        if version != 2:
            raise ValueError("Unsupported ledger version")

        reserved = payload.get("reserved")
        legacy = payload.get("legacy_reserved")
        requests = payload.get("requests")

        if (
            type(reserved) is not int
            or reserved < 0
            or type(legacy) is not int
            or legacy < 0
            or type(requests) is not dict
        ):
            raise ValueError("Invalid ledger structure")

        outstanding = legacy

        for request_id, entry in requests.items():
            if (
                not isinstance(request_id, str)
                or not request_id
                or type(entry) is not dict
                or type(entry.get("credits")) is not int
                or entry["credits"] < 1
                or entry.get("status") not in (
                    "PENDING",
                    "SETTLED",
                    "UNCERTAIN",
                )
            ):
                raise ValueError("Invalid reservation entry")

            if entry["status"] != "SETTLED":
                outstanding += entry["credits"]

        if outstanding != reserved:
            raise ValueError("Reservation accounting mismatch")

        return payload

    except (OSError, ValueError, TypeError) as exc:
        raise ValueError("Invalid reservation ledger") from exc



def _sync_directory(directory):
    """Persist directory-entry changes where supported."""
    fd = os.open(directory, os.O_RDONLY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def _create_recovery_marker(
    path, *, request_id, authorized_at, credits
):
    """Create durable transaction intent before budget reservation."""
    payload = {
        "recovery_required": True,
        "reason": "authorization_commit_uncertain",
        "request_id": request_id,
        "authorized_at": authorized_at,
        "credits": credits,
    }

    # Exclusive creation prevents silently replacing a marker.
    with path.open("x", encoding="utf-8") as handle:
        json.dump(payload, handle)
        handle.flush()
        os.fsync(handle.fileno())

    _sync_directory(path.parent)


def _clear_recovery_marker(path):
    path.unlink()
    _sync_directory(path.parent)

            if any(
                entry.get("status") != "FINALIZED"
                for entry in journal["transactions"].values()
            ):
                raise AuthorizationJournalError(
                    "Authorization blocked: unresolved journal intent"
                )

            decision = check_before_request(
                path=provider_path,
                now=now,
                required_credits=required_credits,
                reserve_credits=reserve_credits,
                max_age_seconds=max_age_seconds,
            )

            ledger = _load_ledger(ledger_path)

            # Cross-check independently persisted authorization
            # evidence before reserving any additional budget.
            journal_transactions = journal["transactions"]
            ledger_requests = ledger["requests"]

            for recorded_id, transaction in journal_transactions.items():
                reservation = ledger_requests.get(recorded_id)

                if reservation is None:
                    raise AuthorizationJournalError(
                        "Authorization blocked: journal-ledger "
                        "request mismatch"
                    )

                if reservation["credits"] != transaction["credits"]:
                    raise AuthorizationJournalError(
                        "Authorization blocked: journal-ledger "
                        "credit mismatch"
                    )

            for recorded_id in ledger_requests:
                if recorded_id not in journal_transactions:
                    raise AuthorizationJournalError(
                        "Authorization blocked: ledger request "
                        "missing from journal"
                    )

            available = (
                decision.remaining_credits - ledger["reserved"]
            )

            if available < required_credits + reserve_credits:
                raise ProviderCreditLimitReached(
                    "Insufficient credits after local reservations"
                )

            # Generate the request ID before any budget mutation.
            request_id = uuid.uuid4().hex

            if request_id in ledger["requests"]:
                raise ValueError("Duplicate request ID")

            # Persist initialization evidence before creating
            # the first authorization intent.
            if not history_path.exists():
                _save_authorization_history(history_path)

            create_intent(
                journal_path,
                request_id=request_id,
                authorized_at=now.astimezone(timezone.utc).isoformat(),
                credits=required_credits,
            )

            # Durable write-ahead intent: never refund automatically.
            _create_recovery_marker(
                recovery_path,
                request_id=request_id,
                authorized_at=now.astimezone(timezone.utc).isoformat(),
                credits=required_credits,
            )

            # Local request quota must succeed before ledger commit.
            reserve_request(
                path=local_budget_path,
                now=now,
            )

            # Read the snapshot timestamp while holding the
            # same lock used by provider-credit refresh.
            provider_payload = json.loads(
                provider_path.read_text()
            )

            snapshot_observed_at = provider_payload["observed_at"]

            # Preserve strict timezone-aware metadata.
            observed = datetime.fromisoformat(
                snapshot_observed_at
            )

            if observed.tzinfo is None:
                raise ValueError(
                    "Provider observation timestamp must be timezone-aware"
                )

            ledger["requests"][request_id] = {
                "credits": required_credits,
                "status": "PENDING",
                "authorized_at": now.astimezone(
                    timezone.utc
                ).isoformat(),
                "snapshot_observed_at": observed.astimezone(
                    timezone.utc
                ).isoformat(),
            }

            ledger["reserved"] += required_credits

            _save_json_atomic(ledger_path, ledger)

            finalize_intent(journal_path, request_id)

            # A failed cleanup leaves authorization blocked,
            # rather than risking an untracked new request.
            _clear_recovery_marker(recovery_path)

            return RequestAuthorization(
                authorized=True,
                provider_credits_remaining=(
                    available - required_credits
                ),
                local_budget_reserved=True,
                request_id=request_id,
            )

        finally:
            fcntl.flock(lock_file.fileno(), fcntl.LOCK_UN)

from strategy.trend_analysis import analyze_trend
from strategy.market_filter import analyze_market_conditions


def generate_signal(
    data,
    expiry_minutes=15,
    data_mode="unknown",
    data_quality_status="INVALID",
):
    """
    Generate an explainable, analysis-only signal.

    Directional analysis is allowed only when:
    - Feed mode is explicitly live or delayed.
    - Data quality status is VALID.
    - At least 50 candles are available.

    Expiry is informational, not a prediction or guarantee.
    """

    if expiry_minutes not in (1, 5, 15):
        raise ValueError("expiry_minutes must be 1, 5, or 15.")

    mode = str(data_mode).strip().lower()
    quality_status = str(data_quality_status).strip().upper()

    def no_trade(reason, trend="UNKNOWN", momentum="UNKNOWN",
                 volatility="UNKNOWN", atr_percent=None):
        return {
            "signal": "NO TRADE",
            "trend": trend,
            "momentum": momentum,
            "volatility": volatility,
            "atr_percent": atr_percent,
            "expiry_minutes": expiry_minutes,
            "reasons": [reason],
        }

    # Safety gate 1: data quality
    if quality_status != "VALID":
        return no_trade(
            f"Data quality is {quality_status}; directional analysis blocked."
        )

    # Safety gate 2: verified feed mode
    if mode not in {"live", "delayed"}:
        return no_trade(
            "Data feed mode is unknown or unsupported. "
            "Verify the provider's feed mode before directional analysis."
        )

    # Safety gate 3: sufficient candles
    if data is None or len(data) < 50:
        return no_trade(
            "At least 50 candles are required."
        )

    row = data.iloc[-1]

    trend_result = analyze_trend(row)
    market_result = analyze_market_conditions(row)

    reasons = [
        trend_result["reason"],
        market_result["reason"],
    ]

    def filtered_no_trade(reason):
        return {
            "signal": "NO TRADE",
            "trend": trend_result["trend"],
            "momentum": market_result["momentum"],
            "volatility": market_result["volatility"],
            "atr_percent": market_result["atr_percent"],
            "expiry_minutes": expiry_minutes,
            "reasons": reasons + [reason],
        }

    if trend_result["trend"] == "UNKNOWN":
        return filtered_no_trade(
            "Trend could not be determined."
        )

    if not market_result["valid"]:
        return filtered_no_trade(
            market_result["reason"]
        )

    trend = trend_result["trend"]
    momentum = market_result["momentum"]

    if trend == "SIDEWAYS":
        return filtered_no_trade(
            "Sideways market: no clear trend."
        )

    if trend != momentum:
        return filtered_no_trade(
            "Trend and momentum are not aligned."
        )

    rsi = float(row["RSI_14"])

    if trend == "BULLISH" and rsi >= 70:
        return filtered_no_trade(
            "RSI is in the overbought zone."
        )

    if trend == "BEARISH" and rsi <= 30:
        return filtered_no_trade(
            "RSI is in the oversold zone."
        )

    # Check conflicting candlestick patterns.
    pattern_direction = str(
        row.get("Pattern_Direction", "NONE")
    ).upper()

    directions = {
        item.strip()
        for item in pattern_direction.split(";")
    }

    if "BULLISH" in directions and "BEARISH" in directions:
        return filtered_no_trade(
            "Conflicting candlestick patterns detected."
        )

    if trend == "BULLISH" and "BEARISH" in directions:
        return filtered_no_trade(
            "Bearish candlestick pattern conflicts with trend."
        )

    if trend == "BEARISH" and "BULLISH" in directions:
        return filtered_no_trade(
            "Bullish candlestick pattern conflicts with trend."
        )

    # Avoid signals immediately next to opposing S/R levels.
    atr = float(row["ATR_14"])
    close = float(row["Close"])

    for column in ("Support", "Resistance"):
        try:
            level = float(row[column])
        except (KeyError, TypeError, ValueError):
            level = None

        if level is not None:
            if trend == "BULLISH" and column == "Resistance":
                if abs(level - close) <= 0.25 * atr:
                    return filtered_no_trade(
                        "Price is too close to resistance."
                    )

            if trend == "BEARISH" and column == "Support":
                if abs(close - level) <= 0.25 * atr:
                    return filtered_no_trade(
                        "Price is too close to support."
                    )

    signal = "UP" if trend == "BULLISH" else "DOWN"

    reasons.append(
        "Trend, momentum and pattern checks are aligned."
    )

    if mode == "delayed":
        reasons.append(
            "WARNING: Analysis uses delayed market data."
        )

    reasons.append(
        "This is an analysis result, not a guaranteed prediction."
    )

    return {
        "signal": signal,
        "trend": trend,
        "momentum": momentum,
        "volatility": market_result["volatility"],
        "atr_percent": market_result["atr_percent"],
        "expiry_minutes": expiry_minutes,
        "reasons": reasons,
    }

"""Research-only market condition classification from completed candles."""
from __future__ import annotations

import math


def _number(value):
    try:
        x = float(value)
    except (TypeError, ValueError):
        return None
    return x if math.isfinite(x) else None


def assess_market(analysis: dict, *, stale: bool = False, gaps: int = 0) -> dict:
    """Describe indicator agreement; never generate a trade signal."""
    base = {
        "condition": "DATA UNAVAILABLE",
        "explanation": "Market assessment unavailable.",
        "ema_view": "Unavailable",
        "rsi_view": "Unavailable",
        "macd_view": "Unavailable",
        "agreement": 0,
        "analysis_only": True,
        "execution_enabled": False,
        "directional_backtesting_allowed": False,
    }
    if stale or gaps:
        base["explanation"] = "Stale candles or timestamp gaps detected; assessment blocked."
        return base
    latest = analysis.get("latest", {})
    if not isinstance(latest, dict):
        return base
    ema9 = _number(latest.get("EMA_9"))
    ema21 = _number(latest.get("EMA_21"))
    rsi = _number(latest.get("RSI_14"))
    if any(x is None for x in (ema9, ema21, rsi)) or not 0 <= rsi <= 100:
        base["explanation"] = "Required indicators are missing or invalid."
        return base

    # MACD column names depend on the existing indicator implementation.
    macd_keys = [k for k in latest if k.upper() == "MACD"]
    signal_keys = [k for k in latest if "SIGNAL" in k.upper() and "MACD" in k.upper()]
    if not signal_keys:
        signal_keys = [k for k in latest if k.upper() == "SIGNAL"]
    macd = _number(latest[macd_keys[0]]) if macd_keys else None
    signal = _number(latest[signal_keys[0]]) if signal_keys else None

    ema_vote = 1 if ema9 > ema21 else (-1 if ema9 < ema21 else 0)
    # RSI > 50 / < 50 reflects relative momentum, not a trade recommendation.
    rsi_vote = 1 if rsi > 50 else (-1 if rsi < 50 else 0)
    macd_vote = (1 if macd > signal else (-1 if macd < signal else 0)) if macd is not None and signal is not None else None

    base["ema_view"] = "EMA 9 above EMA 21" if ema_vote > 0 else "EMA 9 below EMA 21" if ema_vote < 0 else "EMAs equal"
    base["rsi_view"] = f"RSI {rsi:.1f} ({'above' if rsi > 50 else 'below' if rsi < 50 else 'at'} midpoint 50)"
    base["macd_view"] = ("MACD above signal" if macd_vote > 0 else "MACD below signal" if macd_vote < 0 else "MACD equals signal") if macd_vote is not None else "MACD comparison unavailable"

    votes = [ema_vote, rsi_vote] + ([macd_vote] if macd_vote is not None else [])
    positive = sum(v == 1 for v in votes)
    negative = sum(v == -1 for v in votes)
    base["agreement"] = max(positive, negative)
    if positive == len(votes) and len(votes) == 3:
        base["condition"] = "BULLISH ALIGNMENT"
        base["explanation"] = "EMA, RSI and MACD observations align upward; this does not predict the next candle."
    elif negative == len(votes) and len(votes) == 3:
        base["condition"] = "BEARISH ALIGNMENT"
        base["explanation"] = "EMA, RSI and MACD observations align downward; this does not predict the next candle."
    else:
        base["condition"] = "MIXED / INCONCLUSIVE"
        base["explanation"] = "Indicators are mixed or insufficient for a three-indicator agreement assessment."
    return base
